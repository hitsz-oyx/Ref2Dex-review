sample = {
    # ============================================================
    # 1. Bookkeeping / indexing
    # ============================================================

    "seq_id": str,
    # 当前 sample 所属的 GRAB 序列 ID。
    # 用于轨迹分组、数据划分和 debug。

    "frame_id": int,
    # 当前帧编号 t。
    # 该 sample 通常表示从 t 到 t+1 的一步预测样本。

    "dt": float,
    # 当前帧 t 到下一帧 t+1 的时间间隔，单位为秒。
    # 如果需要速度，可以用 flow / dt。


    # ============================================================
    # 2. Object surface points at time t
    # ============================================================

    "obj_points": Float[No, 3],
    # 当前帧 t 下的物体表面采样点，世界坐标系。
    # No 是物体表面点数量。
    # 这些点应来自 canonical object mesh 的固定采样点，
    # 再通过当前帧 obj_root_pose 变换到世界坐标系。

    "obj_normals": Float[No, 3],
    # 当前帧 t 下每个物体采样点的法向，世界坐标系。
    # 对刚体物体，可由 canonical normal 乘以 obj_root_pose 的旋转部分得到。

    "obj_point_id": Int[No],
    # 每个物体采样点的 persistent ID。
    # 同一个 obj_point_id 在所有帧中都表示 canonical object mesh 上的同一个采样点。

    "obj_root_pose": Float[4, 4],
    # 当前帧 t 的物体根位姿。
    # 通常表示 object local frame -> world frame 的 SE(3) 矩阵。


    # ============================================================
    # 3. Hand surface points at time t
    # ============================================================

    "hand_points": Float[Nh, 3],
    # 当前帧 t 下的手部表面采样点，世界坐标系。
    # Nh 是手表面点数量。
    # 这些点应来自 MANO / SMPL-X hand template 上的固定采样点，
    # 再通过当前帧手部 mesh deformation 得到。

    "hand_normals": Float[Nh, 3],
    # 当前帧 t 下每个手部采样点的法向，世界坐标系。
    # 可由 posed hand mesh 的 vertex normal 通过 barycentric interpolation 得到。

    "hand_point_id": Int[Nh],
    # 每个手部采样点的 persistent ID。
    # 同一个 hand_point_id 在所有帧中都表示 canonical hand surface 上的同一个采样点。

    "hand_root_pose": Float[4, 4],
    # 当前帧 t 的手部根位姿。
    # 通常表示 wrist/root frame -> world frame 的 SE(3) 矩阵。
    # 注意：hand_root_pose 只描述手腕整体运动，不描述手指关节运动。


    # ============================================================
    # 4. Hand joint pose at time t
    # ============================================================

    "hand_joint_axis_angle": Float[J, 3],
    # 当前帧 t 的手部关节轴角表示。
    # J 是手部关节数量。
    # 每个关节用一个 3D axis-angle 向量表示：
    #   direction = rotation axis
    #   norm      = rotation angle in radians
    #
    # 对 MANO 来说，常见形式是：
    #   hand_joint_axis_angle: Float[15, 3]
    # 表示除 global wrist/root 外的 15 个手部关节。
    # 如果你把 wrist/global orientation 也包含进来，则可以使用 Float[16, 3]。
    # 但如果已经存了 hand_root_pose，建议这里只存手指内部关节。

    "hand_joint_axis_angle_delta": Float[J, 3],
    # 从当前帧 t 到下一帧 t+1 的手部关节轴角变化。
    # 可以定义为：
    #   hand_joint_axis_angle_delta = hand_joint_axis_angle_{t+1}
    #                               - hand_joint_axis_angle_t
    #
    # 这是一个简单近似，适合作为最小字段。
    # 更严格的旋转差应在 SO(3) 上计算：
    #   R_delta = R_{t+1} @ R_t^{-1}
    # 再转回 axis-angle。
    # 第一版可以先用简单差分，后面需要更精确时再升级。



    # ============================================================
    # 5. Action and prediction target
    # ============================================================

    "hand_flow": Float[Nh, 3],
    # 手部表面点从当前帧 t 到下一帧 t+1 的位移：
    #   hand_flow[j] = hand_points_{t+1}[j] - hand_points_t[j]
    #
    # 这是模型的主要 action / conditioning signal，
    # 表示人手表面如何运动并作用于物体。

    "obj_flow": Float[No, 3],
    # 物体表面点从当前帧 t 到下一帧 t+1 的位移：
    #   obj_flow[i] = obj_points_{t+1}[i] - obj_points_t[i]
    #
    # 这是第一阶段 scene/object flow prediction 的监督目标。

    "obj_flow_valid": Bool[No],
    # 每个物体点的 flow 是否有效。
    # 对 GRAB 中刚体物体，如果 pose 有效，通常可以全为 True。

    "hand_flow_valid": Bool[Nh],
    # 每个手部点的 flow 是否有效。
    # 如果某些帧手部拟合质量差，可以置为 False 或在 loss 中降低权重。


    # ============================================================
    # 6. Nearest hand-object correspondence at time t

    "hand_to_obj_nn_id": Int[Nh],
    # 对每个 hand point，记录当前帧 t 最近的 object point 的 ID 或索引。
    # hand_to_obj_nn_id[j] 表示 hand point j 当前最接近哪个 object point。

    "hand_to_obj_dist": Float[Nh],
    # 对每个 hand point，记录它到最近 object point 的距离。

    # 对每个 object point，记录当前帧 t 最近的 hand point 的 ID 或索引。
    # obj_to_hand_nn_id[i] 表示 object point i 当前最接近哪个 hand point。

    "obj_to_hand_dist": Float[No],
    # 对每个 object point，记录它到最近 hand point 的距离。
    # 可用于生成 object-side soft contact label，
    # 也可用于给接触附近的 object points 加大 scene-flow loss 权重。

    # 7. Optional but recommended hand semantics
    # ============================================================

    "hand_finger_id": Int[Nh],
    # 每个 hand point 所属的手指语义 ID。
    # 例如：
    #   0 = palm
    #   1 = thumb
    #   2 = index
    #   3 = middle
    #   4 = ring
    #   5 = little

    "hand_region_id": Int[Nh],
    # 每个 hand point 所属的局部区域语义 ID。
    # 例如：
    #   0 = palm
    #   1 = fingertip
    #   2 = finger_pad
    #   3 = finger_side
    #   4 = finger_back
    #   5 = hand_back
}
