    # 当前 sample 所属的 GRAB / ARCTIC 序列 ID。

    # 通常表示 object local frame -> world frame 的 SE(3) 矩阵。

    # 3. Hand surface points at time t (face-center 采样 Nh=1538)
    # ============================================================
    # 下面是右手模板;左手用 `left_hand_*` 替换前缀，时间维度统一为 T。

    "right_hand_points": Float[T, 1538, 3],
    # 当前帧 t 下的手部表面采样点，世界坐标系。
    # Nh = 1538，对应 MANO face 数量。
    # 这些点是 MANO hand template 的 face center（每面一个），拓扑稳定。
    # **所有 T 帧都保留**，无接触帧不删除、不清零，使用 right_hand_valid 掩码标记。

    "right_hand_normals": Float[T, 1538, 3],
    # 当前帧 t 下每个手部采样点的法向，世界坐标系。
    # 用 posed hand mesh 的 face normal（与 face center 天然对应）。

    "right_hand_point_id": Int[1538],
    # 每个手部采样点的 persistent ID（拓扑稳定，与姿态无关）。

    "right_hand_root_pose": Float[T, 4, 4],
    # 当前帧 t 的手部根位姿（wrist 节点到 world 的 SE(3)）。

    # 4. Hand full mesh + shape parameters at time t
    # ============================================================

    "right_hand_verts": Float[T, 778, 3],
    # 当前帧 t 的 MANO 完整 778 顶点（世界坐标），无降采样。

    "right_hand_betas": Float[T, 10],
    # MANO 形状参数 β（shape coefficients），长度固定为 10。
    # ARCTIC 数据中每帧的 β 通常相同（一个 subject 一套手型），
    # 但为保持 (T, 10) 形状仍按帧存储。**该字段是手部内在属性**。


    # ============================================================
    # 5. Hand joint pose at time t
    # ============================================================

    "right_hand_joint_axis_angle": Float[T, 15, 3],
    # 当前帧 t 的 15 个手指关节（不含 wrist）axis-angle 表示。
    # 顺序: thumb(0-2), index(3-5), middle(6-8), ring(9-11), pinky(12-14)。

    "right_hand_joint_axis_angle_delta": Float[T, 15, 3],
    # 当前帧 t 的 15 个手指关节 axis-angle 帧间差分
    # (delta[t] = aa[t+1] - aa[t], 最后一帧填 0)。

    "right_hand_joint_quat": Float[T, 16, 4],
    # 当前帧 t 的 16 个 MANO 关节世界空间旋转四元数 [w, x, y, z]。
    # 顺序: 0=wrist(根), 1-3=thumb, 4-6=index, 7-9=middle, 10-12=ring, 13-15=pinky。
    # 计算方式: 沿 MANO 运动学树组合 global_orient + hand_pose，
    #           R_world[j] = R_world[parent[j]] @ R_local[j-1]，再转四元数。
    # 四元数约定: [w, x, y, z]（实部在前），与 PyTorch3D / scipy 兼容。
    # 转换工具: arctic_preprocess.rotmat_to_quat (Shepperd 四分支法，数值稳定)。


    # ============================================================
    # 6. Action and prediction target
    # ============================================================

    "right_hand_flow": Float[T, 1538, 3],
    # 当前帧 t 的手部表面点位移:
    #   flow[t] = points[t+1] - points[t],  t < T-1
    #   flow[T-1] = 0 (最后一帧没有"下一帧")

    "right_hand_flow_valid": Bool[T, 1538],
    # 每个手部点的 flow 是否有效。
    # 受 fitting_err 阈值过滤: fitting_err > 10mm 的帧整帧标记 False；
    # 最后一帧（无下一帧）也标记 False。
    # **该字段与 right_hand_valid 独立**：描述的是拟合质量，不是接触状态。

    "obj_flow": Float[T, No, 3],
    # 物体表面点从 t 到 t+1 的位移 (T 帧, 最后一帧填 0)。

    "obj_flow_valid": Bool[T, No],
    # 每个物体点的 flow 是否有效（对刚体物体一般全 True）。

    # 双向最近邻字段保持 T 帧，与 valid 掩码配合使用。

    "right_hand_to_obj_dist": Float[T, 1538],
    # 当前帧 t: 每个手点到最近物点的距离（米）。
    # **该距离的 min 即用于判断 contact 的判据**。

    "right_hand_min_dist_to_obj": Float[T],
    # 当前帧 t: 右手所有采样点到物体的最近距离（右手的 NN dist 的 min）。
    # 下游可基于此构造 soft contact 软接触 label（如 -exp(-d/τ)）。

    "obj_to_right_hand_nn_id": Int[T, No],
    # 当前帧 t: 每个物点对应的最近手点 ID。

    "obj_to_right_hand_dist": Float[T, No],
    # 当前帧 t: 每个物点到最近手点的距离（米）。


    # ============================================================
    # 8. Hand-object contact validity mask (per-hand, per-frame, 三阈值)
    # ============================================================
    # **核心约定**: 所有手部数据保持 (T, ...) 形状完整保留，无接触帧
    # **不删除、不清零**，仅通过 valid 掩码标记。
    #
    # 三阈值设计 (2026-06):
    #   - *_hand_valid       (3cm): "手在物体附近" — 用于 soft contact 概率化 / 训练加权
    #   - *_hand_valid_2cm   (2cm): "手真的在接触" — 过渡口径，向后兼容
    #   - *_hand_valid_1cm   (1cm): "手紧贴物体" — 与 CPF 内部 407 palm 顶点
    #                                 range_threshold=10mm 严格对齐
    #
    # 3cm 掩码 (默认 thresh = 0.03m, 可通过 CLI 的 --contact_distance_thresh 覆盖):
    #   right_hand_valid[t] = (该手 1538 face-center 点 min dist_to_obj ≤ 0.03m)
    #   left_hand_valid[t]  = (该手 1538 face-center 点 min dist_to_obj ≤ 0.03m)
    # 2cm 掩码 (硬编码 0.02m, 过渡兼容):
    #   right_hand_valid_2cm[t] = (该手 1538 face-center 点 min dist_to_obj ≤ 0.02m)
    #   left_hand_valid_2cm[t]  = (该手 1538 face-center 点 min dist_to_obj ≤ 0.02m)
    # 1cm 掩码 (硬编码 0.01m, 与 CPF range_threshold=10mm 严格对齐):
    #   right_hand_valid_1cm[t] = (该手 1538 face-center 点 min dist_to_obj ≤ 0.01m)
    #   left_hand_valid_1cm[t]  = (该手 1538 face-center 点 min dist_to_obj ≤ 0.01m)
    #
    # 双向独立判断:
    #   右手无接触只影响 right_hand_valid[t]=False（2cm / 1cm 同理）
    #   左手无接触只影响 left_hand_valid[t]=False
    #   双手都无接触时两个 valid 都是 False，但 hand_* 数据仍然保留

    # 下游推荐用法:

    #               fallback 到 *_hand_valid_2cm / *_hand_valid
    #   - 数据加载: 配合 hand_to_obj_dist 做 soft contact 概率化处理
    #   - 物理仿真: 跳过 valid[t]=False 的帧（手在空中，不参与抓取）

    "right_hand_valid": Bool[T],
    # 右手 3cm 接触有效性掩码 (T,)。
    # True  = 该帧该手在接触距离内（≤ 0.03m）
    # False = 该帧该手无接触（> 0.03m）

    "right_hand_valid_2cm": Bool[T],
    # 右手 2cm 接触有效性掩码 (T,)，过渡口径，向后兼容。
    # True  = 该帧该手在硬接触距离内（≤ 0.02m）
    # False = 该帧该手未达硬接触距离（> 0.02m）
    # **与 right_hand_valid 的关系**: valid_2cm 是 valid 的真子集
    # （valid=True 的帧数 ≥ valid_2cm=True 的帧数）。

    "right_hand_valid_1cm": Bool[T],
    # 右手 1cm 接触有效性掩码 (T,)，与 CPF 内部 palm 顶点

    # True  = 该帧该手紧贴物体（≤ 0.01m）
    # False = 该帧该手未紧贴物体（> 0.01m）
    # **与 valid_2cm 的关系**: valid_1cm ⊆ valid_2cm ⊆ valid。
    # **优先使用**: 训练 CPF / 做接触优化时优先用 valid_1cm，
    # 避免 ARCTIC "face-center 离物 2cm / palm 离物 2.2cm" 的边缘帧
    # 被误激活 contact_loss。

    "left_hand_valid": Bool[T],
    # 左手 3cm 接触有效性掩码 (T,)。语义同 right_hand_valid。

    "left_hand_valid_2cm": Bool[T],
    # 左手 2cm 接触有效性掩码 (T,)。语义同 right_hand_valid_2cm。

    "left_hand_valid_1cm": Bool[T],
    # 左手 1cm 接触有效性掩码 (T,)。语义同 right_hand_valid_1cm。


    # ============================================================
    # 9. Optional but recommended hand semantics
    # ============================================================

    "right_hand_finger_id": Int[1538],
    # 每个手部采样点所属的手指语义 ID（拓扑稳定，与姿态无关）。
    # 0=palm, 1=thumb, 2=index, 3=middle, 4=ring, 5=pinky

    "right_hand_region_id": Int[1538],
    # 每个手部采样点所属的局部区域语义 ID（拓扑稳定，与姿态无关）。
    # 0=palm, 1=fingertip, 2=finger_pad, 3=finger_side, 4=finger_back, 5=hand_back

    "right_hand_is_mano": Bool,
    # 该手的 hand_* 字段是否由 MANO forward 拟合得到。
    # - ARCTIC：恒为 True（.mano.npy 提供的 rot/pose/transl/shape 直接走
    #   smplx.MANO forward 得到）。
    # - GRAB：  恒为 True（hand_params 来自 SMPL-X 拟合 + MANO forward + vtemp
    #   注入）。
    # **该字段不写入 .npz**，改为写入 output_root/meta.json 的 mano_meta 字段，
    # 以保持 .npz 与 ARCTIC 字段完全一致。

    "right_hand_has_mano_params": Bool,
    # 原始数据中是否存了该手的 mano params。
    # - True:  原始数据包含该手的手部参数（global_orient, hand_pose, transl, betas）
    # - False: 原始数据没存该手 → 该侧 hand_* 全部为占位零
    # **该字段不写入 .npz**，改为写入 output_root/meta.json 的 mano_meta 字段。

    "left_hand_is_mano": Bool,
    # 语义同 right_hand_is_mano（左手版）。
    # **该字段不写入 .npz**，改为写入 output_root/meta.json 的 mano_meta 字段。

    "left_hand_has_mano_params": Bool,
    # 语义同 right_hand_has_mano_params（左手版）。
    # **该字段不写入 .npz**，改为写入 output_root/meta.json 的 mano_meta 字段。
