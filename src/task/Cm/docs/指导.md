# 3. 加一个简单 Slot Attention

建议直接放进 `src/task/Cm/model.py`。

```python
class SlotAttention(nn.Module):
    """Compress dense hand-action features U_h into K sparse Cm slots."""

    def __init__(
        self,
        dim: int,
        num_slots: int,
        num_iters: int = 3,
        hidden_dim: int | None = None,
    ) -> None:
        super().__init__()

        self.dim = int(dim)
        self.num_slots = int(num_slots)
        self.num_iters = int(num_iters)
        hidden_dim = hidden_dim or dim * 2

        # K distinct learned initial slots.
        # Do not initialize all slots identically, otherwise symmetry can cause collapse.
        self.slot_init = nn.Parameter(
            torch.randn(1, self.num_slots, self.dim) * 0.02
        )

        self.norm_input = nn.LayerNorm(dim)
        self.norm_slots = nn.LayerNorm(dim)
        self.norm_mlp = nn.LayerNorm(dim)

        self.to_q = nn.Linear(dim, dim, bias=False)
        self.to_k = nn.Linear(dim, dim, bias=False)
        self.to_v = nn.Linear(dim, dim, bias=False)

        self.gru = nn.GRUCell(dim, dim)

        self.mlp = nn.Sequential(
            nn.Linear(dim, hidden_dim),
            nn.GELU(),
            nn.Linear(hidden_dim, dim),
        )

        self.scale = dim ** -0.5
        self.eps = 1e-8

    def forward(
        self,
        inputs: torch.Tensor,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Args:
            inputs: [B, N_hand, D]

        Returns:
            slots: [B, K, D]
            weights: [B, K, N_hand]
                Final normalized soft assignment used to aggregate hand points.
        """
        batch_size = inputs.shape[0]

        inputs = self.norm_input(inputs)
        k = self.to_k(inputs)  # [B, N, D]
        v = self.to_v(inputs)  # [B, N, D]

        slots = self.slot_init.expand(batch_size, -1, -1)

        weights = None

        for _ in range(self.num_iters):
            slots_prev = slots

            q = self.to_q(self.norm_slots(slots))
            q = q * self.scale  # [B, K, D]

            # [B, K, N]
            logits = torch.einsum("bkd,bnd->bkn", q, k)

            # Important:
            # each hand point competes across K slots.
            assignment = torch.softmax(logits, dim=1)
            assignment = assignment + self.eps

            # Normalize over hand points when computing each slot update.
            weights = assignment / assignment.sum(
                dim=-1, keepdim=True
            ).clamp_min(self.eps)

            # [B, K, D]
            updates = torch.einsum("bkn,bnd->bkd", weights, v)

            slots = self.gru(
                updates.reshape(-1, self.dim),
                slots_prev.reshape(-1, self.dim),
            ).reshape(batch_size, self.num_slots, self.dim)

            slots = slots + self.mlp(self.norm_mlp(slots))

        assert weights is not None
        return slots, weights
```

这里：

[
U^h\in\mathbb R^{B\times1538\times256}
]

经过：

[
\operatorname{SlotAttn}(U^h)
]

得到：

[
C_m\in\mathbb R^{B\times K\times256}
]

以及：

[
A\in\mathbb R^{B\times K\times1538}
]

其中 (A_{kj}) 表示 slot (k) 从第 (j) 个 hand-action feature 聚合多少信息。

# 6. 用 Slot assignment 计算 Cm 的几何属性

以前是 hard gather 一个 MANO face：

[
m_k=y_{\operatorname{TopK}(k)}
]

现在改成 soft anchor：

[
m_k
===

\sum_jA_{kj}y_j
]

代码：

```python
cm_anchor_pos = torch.einsum(
    "bkn,bnd->bkd",
    cm_assignment,
    hand_points,
)

cm_hand_flow = torch.einsum(
    "bkn,bnd->bkd",
    cm_assignment,
    hand_flow,
)

cm_anchor_normal = torch.einsum(
    "bkn,bnd->bkd",
    cm_assignment,
    hand_normals,
)

cm_anchor_normal = torch.nn.functional.normalize(
    cm_anchor_normal,
    dim=-1,
    eps=1e-6,
)
```

于是每个 slot 有：

[
C_{m,k}
=======

(
c_k,,
m_k,,
\Delta m_k,,
n_k
)
]

其中：

* `cm_tokens[k]`：动作 latent；
* `cm_anchor_pos[k]`：该动作组在手上的软中心；
* `cm_hand_flow[k]`：该组平均动作位移；
* `cm_anchor_normal[k]`：该组平均表面朝向。

不再需要：

```python
cm_anchor_idx
```

因为一个 slot 不再对应某一个离散 MANO face，而是一个**手表面 action region**。

            Z_h                         Z_o
             │                           │
             │                           └─ [Z_o, x_o, n_o]

             │                               object_context
             │
             ├─ y_h
             ├─ n_h

                  │
                  ↓

## 第一版不要加的东西

先全部不要：

* `activity_head`
* hard `TopK`
* `cm_active`
* activity loss
* (P_t) guidance
* slot diversity loss
* slot sparsity loss
* `motion_to_scene` global cross-attention

因为 `Z_h` 已经是从联合 ((O_t,H_t)) PTv3 得到的 scene-conditioned dense token；当前 frozen encoder 的确联合编码 object 与 hand。

第一轮实验只回答一个问题：

[
\boxed{
\text{完整版 }U^h
\overset{\text{Slot Attention}}{\longrightarrow}
16\text{ 个 }C_m
\text{，是否足以重建 object point flow？}
}
]

这个版本一旦能 overfit，再看 `cm_assignment` 是否自然落在拇指、指腹、掌面等功能区域；之后才值得考虑加入 (P_t) 或 slot regularization。
